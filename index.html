⚙️ ตั้งค่าเชื่อมต่อ Google Apps Script Web App

กำหนด Web App URL เพื่อส่งข้อมูลไปบันทึกลงใน Google Sheets ของคณะฯ อัตโนมัติ

📌 ขั้นตอนการตั้งค่า: นำโค้ด Google Apps Script (`doPost`) ไปวางใน Apps Script ของ Google Sheet จากนั้นคลิก Deploy > New deployment > Web App กำหนดสิทธิ์ผู้เข้าถึงเป็น Anyone (ทุกคน) แล้วคัดลอก URL มาวางในช่องนี้

✅ แจ้งเตือนสำเร็จ
Google Apps Script Code (Code.gs): 
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    
    // หากเป็นชีตใหม่และยังไม่มีหัวข้อ ให้สร้าง Header
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        "Timestamp", 
        "เลขที่ขออนุมัติ", 
        "กำหนดส่งเอกสาร", 
        "ผู้ขออนุมัติ", 
        "ผู้ตรวจเอกสาร", 
        "ประเภทค่าใช้จ่าย", 
        "รายการเอกสารที่ติ๊กเลือก", 
        "หมายเหตุเจ้าหน้าที่"
      ]);
      sheet.getRange("1:1").setFontWeight("bold").setBackground("#EDE7F6");
    }

    var data = JSON.parse(e.postData.contents);
    
    sheet.appendRow([
      data.timestamp || new Date(),
      data.approvalNo || "",
      data.dueDate || "",
      data.requester || "",
      data.checker || "",
      data.expenseTypes || "",
      data.checkedDocuments || "",
      data.officerNote || ""
    ]);

    return ContentService.createTextOutput(JSON.stringify({ "status": "success" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ "status": "error", "message": err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
